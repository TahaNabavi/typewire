@GrpcEndpoint(contracts.user.syncUser) // POST /user.v1.UserService/GetUser
async syncUser(@ContractInput() input) {
  const user = await this.users.byId(input.id);
  if (!user) throw new GrpcException(GrpcCode.NotFound);
  return user; // ✓ checked against `response`
}
